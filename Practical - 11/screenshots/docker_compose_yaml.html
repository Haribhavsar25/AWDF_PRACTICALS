📄 docker-compose.yml
🐳 backend/Dockerfile
🐳 frontend/Dockerfile
🚫 .dockerignore
version: '3.8'

services:
# 1. Official MongoDB Container with Persistent Named Volume
mongodb:
image: mongo:6.0
container_name: awdf-mongodb
restart: always
ports:
- "27017:27017"
volumes:
- mongo-data:/data/db
networks:
- app-network

# 2. Node.js Express REST API Container
backend:
build:
context: ./backend
dockerfile: Dockerfile
container_name: awdf-backend
ports:
- "5000:5000"
environment:
- PORT=5000
- MONGO_URI=mongodb://mongodb:27017/taskdb
depends_on:
- mongodb
networks:
- app-network

# 3. React Frontend Multi-Stage Container
frontend:
build:
context: ./frontend
dockerfile: Dockerfile
container_name: awdf-frontend
ports:
- "5173:5173"
depends_on:
- backend
networks:
- app-network
